\lecture{2}{Mon 13 Feb 2023 16:45}{Functions of bounded variation}

Let $f: [a,b] \to  \mathbb{R}$. Let
\[
	a \le  x_0 \le  x_1 \le  \ldots \le  x_k \le b
.\] 
and define
\[
	T(f, x_0, \ldots, x_k) = \sum_{i=1}^{k} \left| f(x_i) - f(x_{i-1}) \right| 
.\] 
be the \textit{total variation} of $f$.

Let $T(f) = T_a^b(f) = \operatorname{sup}_{a\le x_i\le b}T(f, x_0, \ldots, x_k) \le \infty $

\begin{definition}[bounded variation]
	$f$ is of \textit{bounded variation} if $T(f) < \infty $, we write $f \in \text{BV}[a,b]$.
\end{definition}

\begin{itemize}
	\item 
		\begin{example}
			if $f$ increases, then
			\[
				T(f, x_0, \ldots, x_k) = f(x_k) - f(x_0) \le f(b) - f(a) = T(f, a, b)
			.\] 
			Hence monotone functions are of BV.
		\end{example}
	\item 
		\begin{example}
			Let $D(x) = \mathbf{1}_{\mathbb{Q}}$. If $x_0<x_1<\ldots<x_k$ alternate between rationals an irrationals, then $T(D) = \infty $.
		\end{example}
	\item 
		\begin{example}
			Define a function of unbounded oscillation:
			\[
				g(x) = \begin{cases}
					0 & x = 0,  \frac{1}{k} \\
					x & \frac{1}{2}(\frac{1}{k} + \frac{1}{k+1})\\
					\text{linear in between}
				\end{cases}
			.\] 
			Then $g$ is a continuous functions $[0,1] \to \mathbb{R}$.
			Let $x_0 = \frac{1}{p}, x_2=\frac{1}{p-1 },\ldots, x_{2(p-1)}=1$.
			Now
			\[
				T(g, x_0, \ldots, x_{2(p-1)} \ge  1  + \frac{1}{2} + \frac{1}{3} + \ldots \to  \infty 
			.\] 
			Hence continuous functions may not be of BV.

		\end{example}
\end{itemize}

\begin{theorem}(Characterization of functions of BV, Jordan)
	Let $f$ be real-valued on $[a,b]$. Then $f \in BV$ if and only if there exists real-valued $g, h$ defined on  $[a,b]$ where $g,h$ are increasing and $f = g - h$.
\end{theorem}
\begin{proof}
	($\impliedby$) By HW, $T(f) = T(g-h) \le T(g) + T(h) < \infty $ 

	($\implies$) Let $g(x) = T_a^x (f)$ and $h(x) = T_a^x(f) - f(x)$, then  $g-h = f$. It remains to show $g, h$ increasing.

	If $a\le x\le y\le b$ then
	\[
		g(y) = T_a^y (f) = T_a^x(f) + T_x^y (f) \ge g(x)
	.\] 
	\[
		h(y) = T_a^y(f) - f(y) = T_a^x (f) - f(x) + T_x^y (f) - \left( f(y) - f(x) \right) \ge h(x)
	.\] 
\end{proof}

\begin{theorem}[Dini's Theorem]
	Let $M$ be a compact metric space, $f_n: M \to \mathbb{R}$ be u.s.c. At each $x \in M$ the sequence $f_n(x)$ is monotone increasing, and $f_n \to f$ pointwise where $f$ continuous. Then $f_n \to f$ uniformly.
\end{theorem}
