\subsubsection{Riemann Lebesgue Theorem}



\begin{theorem}[Riemann-Lebesgue Theorem]
	If a function $f$ is bounded on $[a,b]$, then it is Riemann integrable iff its is continuous a.e.
\end{theorem}
Note that Riemann integrability requires boundedness.
\begin{proof}
	($\implies$) Assume $f$ is Riemann integrable. Then there exists a sequence of elementary functions $g_n$ monotone increasing, such that $g_n \le  f$ and 
$R.\int g_n \to R. \int f$. Similarly, there exists elementary functions $h_n \ge f$, monotone decreasing and $R.\int  h_n \to R. \int f$. Both $g_n$ and $h_n$ have limits, call them $g$ and $h$. Then $g \le h$, so $g = h$ a.e. 

Now, take some $x \in [a,b]$ such that $g(x) = h(x) $ and all $g_n, h_n$ are continuous at $x$. Then $f$ is continuous at $x$. Since $g_n, h_n$ each discontinuous at finitely many points, almost every $x \in [a,b]$ satisfies the above condition. Therefore $f$ continuous a.e.

	($\impliedby$) If $P: a = x_0 < x_1 < \ldots < x_n = b$, let
	\[
		l_P(x) = \begin{cases}
			\inf_{x_{j-1}, x_j} f & x \in (x_{j-1}, x_j)\\
			f(x_j) & x = x_j
		\end{cases}
	.\] 
\[
		u_P(x) = \begin{cases}
			\sup_{x_{j-1}, x_j} f & x \in (x_{j-1}, x_j)\\
			f(x_j) & x = x_j
		\end{cases}
	.\]
	For any partition $P$, let
	\[
		\delta(P) = \max_{j = 1\ldots, n} (x_j - x_{j-1})
	.\] 
	We claim: If $P_k$ are partitions, $k \in \mathbb{N}$, such that $\delta(P_k) \to 0$, and $f$ is continuous at $c \in [a,b]$, then $l_{P_k}(c) \to  f(c)$ and $u_{P_k}(c) \to f(c)$.

	Now we verify the claim. Let $\varepsilon>0$. Then there exists $\delta>0$ such that $\left| x - c \right| < \delta$ implies $\left| f(x) - f(c) \right| <\varepsilon$. Choose $k_0$ such that $\delta(P_k) < \delta$ as $k > k_0$. Take such $k$, say $P_k: a = x_0 < x_1 < \ldots < x_n = b$. Let $c \in [x_{j-1}, x_j]$, then $l_{P_k}(c) \ge f(c) - \varepsilon$. Similarly $u_{P_k} \le f(c) +\varepsilon$. Combining the inequalities we get the desired convergence. 

	So
	\[
		L_{P_k} = \int _{[a,b]} l_{P_k} \to \int _{[a,b]} f
	.\] 
	and similarly $U_{P_k} \to  \int _{[a,b]}f$.

	Therefore
	\[
		\sup_P L_P \ge \lim_{k \to \infty }L_{P_k} = \lim_{k \to  \infty } U_{P_k} \ge \inf_P U_P \ge \sup _P L_P
	.\] 
	Hence everything is equal, so the lower and upper integral matches, so $f$ is integrable.
\end{proof}

\begin{example}
	
For example of a function continuous a.e., consider some $f : \mathbb{R}\to \mathbb{R}$ such that
\[
	f\left(\frac{m}{n}\right) = \frac{1}{n} \text{ for } m \in \mathbb{Z}, n \in \mathbb{N}, \text{gcd}(m, n) = 1
.\] 
And $f = 0$ on $\mathbb{R}\setminus \mathbb{Q}$. Then $f$ is continuous at every irrational number.
\end{example}

\subsection{Improper Integral}

\begin{proposition}
	If $h: [a, \infty ) \to  [0,\infty )$ has improper integral $\int _a^{\infty }h$, then $h$ has Lebesgue integral
	\[
		\int _{[a, \infty  )} h = \int _a^\infty  h
	.\] 
\end{proposition}
\begin{proof}
	\[
		\int _a^b h = \int _{[a,b]}h = \int _{[a, \infty )} h \chi_{[a,b]}
	.\] 
	As $b \in \mathbb{N}$ increases, $h \chi_{[a,b]}$ increases pointwise. By MCT, 
	\[
		\int _{[a, \infty )} h \chi_{[a,b]}\to  \int _{[a, \infty )} h
	.\] 
	But by assumption
	\[
		\int _a^b h \to  \int _a^{\infty }h
	.\] 

\end{proof}
