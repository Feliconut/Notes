	\item 8. If every right coset of $H$ in $G$ is a left coset of $H$ in $G$, prove that $a H a^{-1}=H$ for all $a \in G$
\begin{proof}
	We interpret the assumption as: For any right coset $Ha$ in $G$,  $Ha = aH$. Note that this also implies every left coset of $H$ in $G$ is a right coset. Now

	\begin{align*}
		x &\in aHa^{-1} \\
		  &\iff x \in a (Ha^{-1} )\\
		  &\iff x \in a (a^{-1} H)\\
		  &\iff x \in (aa^{-1} )H \\
		  &\iff x \in H
	.\end{align*}
\end{proof}

	\item 16. If $G$ is a finite abelian group and $a_1, \ldots, a_n$ are all its elements, show that $x=a_1 a_2 \cdots a_n$ must satisfy $x^2=e$.
\begin{proof}
	Without loss of generality we assume $a_1, \ldots, a_n$ are distinct.
	Let $a_1'a_2'\ldots a_n'$ be a permutation of $a_1a_2\ldots a_n$. Because $G$ is finite and abelian, permutation does not change the product. Hence $x^2 = (a_1a_2\ldots a_n)(a_1a_2 \ldots a_n) = (a_1a_2\ldots a_n)(a_1'a_2' \ldots a_n')$.

	Now by the inverse axiom, there must exist some $a \in G$ such that $a_1 a = e$. Without loss of generality we can call it $a_1'$. Hence
\[
	(a_1a_2 \ldots a_n) (a_1'a_2' \ldots a_n') = 
	(a_1a_1')(a_2\ldots a_n) (a_2' \ldots a_n') = 
	(a_2 \ldots a_n) (a_2' \ldots a_n')
.\] 
Then we consider the inverse of $a_2$. It cannot be $a_1'$, otherwise $a_2 = \left( a_1' \right) ^{-1} = \left( a_1^{-1} \right) ^{-1} = a_1$, a contradiction. Without loss of generality we may call its inverse $a_2' = a_2^{-1}$. Now
\[
	(a_2 \ldots a_n) (a_2' \ldots a_n') = 
	(a_2 a_2') (a_3 \ldots a_n) (a_3' \ldots a_n')= 
	(a_3 \ldots a_n) (a_3' \ldots a_n')
.\] 
Continue in this fashion we can cancel all terms and onclude that $x^2 = e$.
\end{proof}
\begin{proof}
	(simpler)

	The inverse  \begin{align*}
		i: G &\longrightarrow G \\
		a &\longmapsto i(a) = a^{-1}
	\end{align*} is a bijection. 
	Since $G$ is abelian,
	\[
		(a_1a_2\ldots a_n) =
		(a_1^{-1} a_2^{-1} \ldots a_n^{-1})
	.\] 

	Now
	\begin{align*}
		x^2 &= (a_1a_2\ldots a_n)(a_1a_2\ldots a_n)\\
		&= (a_1a_2\ldots a_n) (a_1^{-1} a_2^{-1} \ldots a_n^{-1})\\
		&= (a_1a_1^{-1})\ldots(a_n a_n^{-1}) \\
		&= e
	.\end{align*} 

\end{proof}

	\item 18. Using the results of Problems 15 and 16 , prove that if $p$ is an odd prime number, then $(p-1) ! \equiv-1 \bmod p$. (This is known as Wilson's Theorem.) It is, of course, also true if $p=2$.
\begin{proof}
	Consider the group $U_p$. Since $p$ is prime, the elements of the group are exactly the natural numbers from $1$ up to $p -1$. Now
	\[
		(p-1)! = 1 \cdot 2 \cdot \ldots \cdot (p-1) =  a_1 a_2 \ldots a_n = x
	.\] 
By Exercise 16, $x^2 = 1$. By Exercise 15, any member $y$ of  $U_p$ that is not $1$ or $p -1$ does not satisfy $y^2 \equiv 1 \pmod p$. Note that: \[
	y^2 \equiv 1 \pmod p \iff y^2 = e \iff y = y^{-1} 
.\] 
Therefore all members except $1 $ or $p -1$ are not their own inverse. Hence they must be pairs that are inverse of each other. Because $U_p$ has even order, and there are even number of elements excluding $1$ and $p -1$, the remaining elements can all pair up correctly. Because  $U_p$ is abelian, we may group the pairs together and cancel them out. Now
\[
	(p-1)! = 1 \cdot (p-1) \equiv -1 \pmod p
.\] 
	
\end{proof}

	\item 27. If $a H=b H$ forces $H a=H b$ in $G$, show that $a H a^{-1}=H$ for every $a \in G$.
\begin{proof}
	Note that 
	\begin{align*}
		& \quad a H = b H \\
		& \iff a^{-1} b \in H  & \text{by definition of left coset}\\
		& \iff b^{-1}  a \in H
	.\end{align*}
	Using the assumption we also have
	\[
		\quad H a = H b 
		\iff a b^{-1}  \in H
		.
	\]
	We have proved that for all $a, b \in G$, \[
		b^{-1} a \in H \implies a b^{-1}  \in H
	.\] 
	Therefore, for all $a, b \in G$, 
	\[
		a b \in H \iff b a \in H
	.\] 
	Fix any $a \in G$. We pick arbitrary $h \in H$ and let $b = a h$. Now $a^{-1} b = a^{-1} a h = h \in H$. Hence $b a^{-1} \in H$. Since $a(a^{-1} b) \in a H$, we have
	\begin{align*}
		a(a^{-1} b)
		&= (a a^{-1} ) b \\
		&= b \\
		&= b (a^{-1} a) \\
		&= (b a^{-1} ) a \\
		&\in H a
	.\end{align*}
	Since $h$ is picked arbitrarily, $b$ can take all elements from $aH$. Since $b \in Ha$, we have shown that $aH \subset  Ha$. Apply the same argument to the other side symmetrically, we obtain \[
		aH = Ha
	\] 
	which means $aHa^{-1} = H$.
\end{proof}

	\item 31. If $o(a)=m$ and $a^s=e$, prove that $m \mid s$.
\begin{proof}
	We first introduce
\begin{lemma}
	For a cyclic group with order $n$ and generating element $a$, we have for all integers $q$ and $r$, \[
		a^q = a^r \iff q \equiv r \pmod n
	.\] 
\end{lemma}
\begin{proof}[Proof of lemma]
	\begin{align*}
		a^q = a^r &\iff (a^q)(a^r)^{-1} = e \\
			  &\iff a^{q-r} = e
	.\end{align*}
	Now we apply euclidean's algorithm to $q-r$, and obtain $x$ and $y$ such that $0\le y<n$, and
	\[
		q-r = n x + y
	.\] 
	Now 
	\[
		a^{q-r} = a^{n x + y} =  \left( a^n \right) ^x a^y = e^x a^y = a^y = e
	.\] 
	Since $0\le y<n$, by properties of cyclic group, $y = 0$(All elements $e, a^1, \ldots, a^n$ must be distinct, so $a^y =  e $ implies $y =0$). Hence
	\[
		q - r = n x \equiv 0 \pmod n
	.\] 

	Now we prove the other direction, and assume $q \equiv r \pmod n$. Apply euclidean algorithm to $q - r$. We obtain integers $x, y$ such that $0\le y < n$  and \[
		q - r = n x + y
	.\] 
	We have $q-r \equiv y \pmod n$. Since $q -r \equiv 0 \pmod n$, $y \equiv 0 \pmod n$. Because it is assumed that  $0\le y<n$, we have $y = 0$. Hence $q - r = nx$. Now
	\begin{align*}
		a^{q-r} &= a^{nx}\\
		&= (a^n)^x \\
		&= e^x \\
			&= e 
	.\end{align*}
	Hence $a^q = a^r $.
\end{proof}

By assumption we have $o(a) = |\left\langle a \right\rangle | = m$. By Theorem 2.4.5, $a^m = e$. Apply the last lemma. Since $a^s = e = a^m$, \[
	s \equiv m \pmod m
.\] 
Hence
\[
	s \equiv 0 \pmod m
.\] 
Which implies $m | s$.
\end{proof}

	\item 37. In a cyclic group of order $n$, show that for each positive integer $m$ that divides $n$ (including $m=1$ and $m=n$ ) there are $\varphi(m)$ elements of order $m$.
\begin{proof}
	An element $a^k \in \left\langle a \right\rangle = G$ has order $m$ iff
	\begin{enumerate}
		\item $(a^k)^m = a^{km} = e$. In other words, $n | k m$, and
		\item $m$ is the smallest positive integer such that $ km $ is a multiple of $n$. In other words, for any $p < m$, we have $n \nmid kp$.
	\end{enumerate}
	Note that $1\le m\le n$ (since $m$ divides $n $), and $0\le k\le n-1$ (since $a^k$ are distinct elements of $G$).

	We introduce a claim that is central to the proof , which we prove later:
	\begin{claim}
		The two conditions above are met, if and only if \[
			(k,n) = \frac{n}{m}
		.\] 
	\end{claim}

	Note that
	\[
			(k,n) = \frac{n}{m} \iff
		\left( \frac{k}{n / m}, \frac{n}{n / m} \right) = \left( \frac{k}{n / m},m \right) = 1 
	.\] 
	Hence $o(a^k) = m \iff \left(\frac{k}{n / m},m\right) = 1$. Since $k < n$, $\frac{k}{n / m}< m$. 
	There are precisely $\varphi(m)$ choices of $\frac{k}{n / m}$ such that $\left(\frac{k}{n / m},m\right) = 1$ and $\frac{k}{n / m}< m$. Each choice of $\frac{k}{n / m}$ then uniquely corresponds to a choice of $k$.
\end{proof}

	\item 2. Recall that $G \simeq G^{\prime}$ means that $G$ is isomorphic to $G^{\prime}$. Prove that for all groups $G_1, G_2, G_3$ :\\
(a) $G_1 \simeq G_1$.\\
(b) $G_1 \simeq G_2$ implies that $G_2 \simeq G_1$.\\
(c) $G_1 \simeq G_2, G_2 \simeq G_3$ implies that $G_1 \simeq G_3$.
\begin{proof}
	\begin{enumerate}
		\item Take the identity mapping $\iota_G: G \to  G$. Now $\iota$ is clearly bijective, and \[
			\iota_G(a_1 a_2) = a_1 a_2 = \iota_G(a_1)\iota(a_2)
		.\] 
		\item Since $G_1 \simeq G_2$, there exists bijective mapping $\varphi: G_1 \to  G_2$ such that for all $a, b \in G_1$, $\varphi(ab) = \varphi(a) \varphi(b)$. Now consider the inverse mapping $\varphi^{-1}$. We know that a bijective mapping has a bijective inverse, so $\varphi^{-1}$ is bijective. For all elements $x, y \in G_2$, let  $a = \varphi^{-1} (x)$ and $b = \varphi^{-1} (y)$. Now $\varphi(a) = x$ and $\varphi(b) = y$. Now \begin{align*}
		\varphi^{-1} (xy) &= \varphi^{-1} (\varphi(a)\varphi(b)) \\
				  &= \varphi^{-1} (\varphi(ab)) \\
				  &= ab \\
				  &= \varphi^{-1} (\varphi(a)) \varphi^{-1}(\varphi(b)) \\
				  &= \varphi^{-1} (x) \varphi^{-1} (y)
		.\end{align*}

	\end{enumerate}

		\[
			\varphi(a)(\varphi(a^{-1} ))=(\varphi(a^{-1} ))\varphi(a)=\varphi(a a^{-1} ) = \varphi(e)
		.\] 
		Hence $\varphi(a^{-1})$ is a inverse for $\varphi(a)$.
	\end{enumerate}
\end{proof}
\begin{proof}
	(Simpler). Take $\varphi(a), \varphi(b) \in \varphi(G)$. We want to show $\varphi(a)\varphi(b)^{-1} \in \varphi(G)$. It suffices to show $\varphi(b)^{-1} = \varphi(b^{-1} )$, then
	\[
		\varphi(a)\varphi(b)^{-1} = \varphi(a)\varphi(b^{-1} ) = \varphi(ab^{-1} ) \in \varphi(G)
	.\] 

	To show this, note that \[
		\varphi(b) \varphi(b^{-1} ) = \varphi(b b^{-1} ) = \varphi(e)
	.\] 
	Also, $\varphi(e)$ is the identity for $\varphi(G)$ since for any $c \in G$, 
	\[
		\varphi(e)\varphi(c) = \varphi(ec) = \varphi(c) = \varphi(ce) = \varphi(c)\varphi(e)
	.\] 
\end{proof}


\item 7. Show that $\varphi: G \rightarrow G^{\prime}$, where $\varphi$ is a homomorphism, is a monomorphism if and only if $\operatorname{Ker} \varphi=(e)$.
\begin{proof}
	Let $e, e'$ be identity of $G$ and $G'$ respectively. If $\varphi$ is a monomorphism, then the mapping is injective. For any element $a \in G$ such that $a \neq e$, we must have $\varphi(a) \neq  \varphi(e) = e'$, hence $a \not\in \operatorname{Ker} \varphi$.

	If $\varphi$ is not a monomorphism, then there exists two distinct elements $a, b \in G$ such that $\varphi(a) = \varphi(b)$. Now $\varphi(ba^{-1}) = \varphi(b)\varphi(a^{-1} ) = \varphi(a)\varphi(a^{-1} )=\varphi(a a^{-1}) = \varphi(e) = e'$. But $ba^{-1} \neq e$ since $a \neq  b$. We have constructed a nonidentity element that is also in $\operatorname{Ker} \varphi$.
\end{proof}

\end{enumerate}
