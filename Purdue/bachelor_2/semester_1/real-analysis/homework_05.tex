\item 12.C. If $C \subseteq \mathbf{R}^{\mathrm{p}}$ is connected, show that its closure $C^{-}$(see Exercise $\left.9 . \mathrm{L}\right)$ is also connected.
\begin{proof}
	Assume $C$ is connected, and suppose $\overline{C} $ is not connected. Take $A,B$ to be a disconnection of $\overline{C} $. The following properties hold:
	\begin{itemize}
		\item $\overline{C}  \subset A \cup B$
		\item $A \cap \overline{C}  \neq \emptyset $
		\item $B \cap \overline{C}  \neq \emptyset $
		\item $(A \cap \overline{C} ) \cap (B \cap \overline{C} ) = \emptyset $
	\end{itemize}

	Moreover, we cannot have both $A \cap C\neq \emptyset $ and $B \cap C \neq \emptyset $, otherwise $C$ is disconnected, since $C \subset A \cup B$ (since $C \subset \overline{C} $ ), and $(A \cap C) \cap (B \cap C) = \emptyset $, a contradiction.

	Without loss of generality, assume $A \cap C = \emptyset $. Then $C \subset B$ since $C \subset A \cup B$. But $A \cap \overline{C} \neq \emptyset $, so there exists $x \in A \cap \overline{C} $ such that $x \in A $ is a cluster point of $C$. However, since $A$ is open, there exists a neighborhood $N$ such that $x \in N \subset A$. Thus $N \cap C = \emptyset $, so $x$ is not a cluster point of $C$, a contradiction.
\end{proof}

\item 12.I. Show that the set
\[
S=\left\{(x, y) \in \mathbf{R}^2: y=\sin \frac{1}{x}, x \neq 0\right\} \cup\{(0, y):-1 \leq y \leq 1\}
\]
is connected in $\boldsymbol{R}^2$. However, it is not always possible to join two points in $S$ by a polygonal curve (or any "continuous" curve) lying entirely in S.
\begin{proof}
	First we prove a lemma:
\begin{lemma}
	If $A \cup B$ is connected and $B \cup C$ is connected, then $A \cup B \cup C$ is connected.

\begin{proof}
	Suppose not. Then $X,Y$ forms a disconnection of $A \cup B \cup C$. Now we cannot have  $(A \cup B)\cap X\neq \emptyset $ and $(A \cup B) \cap Y \neq \emptyset $ simultaneously, otherwise $X,Y$ will form a disconnection of $A \cup B$, a contradiction. Therefore we can assume without loss of generality of $A \cup B \subset X$ and $(A \cup B) \cap Y = \emptyset $.

	Also, we cannot have $(B \cup C) \cap X \neq \emptyset $ and $(B \cup C) \cap Y \neq \emptyset $. Since we already have $(B \cup C) \cap X \neq \emptyset $, we must have $(B \cup C) \cap Y = \emptyset $. Now $(A \cup B \cup C) \cap Y = \emptyset $, so $X,Y  $ is not a disconnection, which is a contradiction.
\end{proof}
\end{lemma}

	We still have not rigorously defined $\sin$ and verified its properties. So I will use the properties of $\sin$ implicitly in the proof without justification.

	First of all,  $\sin$ is a continuous function and $x \mapsto \frac{1}{x}$ is a continuous function. The function $f: \mathbb{R}\setminus \{0\}  \mapsto \mathbb{R}$ such that $f(x) = \sin \frac{1}{x}$ is their composition so is also continuous. Since $\mathbb{R}^+$ is continuous, and $\mathbb{R}^+$ is connected, the set \[
		S_1 = \{(x,y) \in \mathbb{R}^2: y = \sin \frac{1}{x}, x > 0\} 
	\] 
	is connected, since it is the graph of a continuous function over a connected region.

	By symmetry, the set \[
		S_2 = \{(x,y) \in \mathbb{R}^2: y = \sin \frac{1}{x}, x < 0\} 
	\] 
	is connected. Now let 
	\[
		S_3 = \{(0,y): -1 \le y \le 1\} 
	.\] Note that
	\[
		\overline{S_1} = S_1 \cup S_3 
	.\] 
	To justify this, note that $0$ is an oscillating discontinuity of $f$. This means $f$ oscillates near $0$ infinitely often. More precisely, for any $y \in [-1,1]$ and $\delta > 0$, there exists some $0 < x < \delta$ such that $f(x) = y$. Using this property, we can conclude that every point $(0,y)$ such taht $y \in [0,1]$ is a cluster point of $S_1$.
	
	By the last exercise,  $\overline{S_1} = S_1 \cup S_3$ is connected. Symmetrically, $\overline{S_2}  = S_2 \cup S_3$ is connected. Hence $S = S_1 \cup S_2 \cup S_3$ is connected.

	Consider $s_1 = (0, 1)$ and $s_2 = (1, \sin 1)$. For a curve $\gamma: [0,1] \to [-1, 1]$ such that $\gamma(0) = s_1$ and $\gamma(1) = s_2$ to be entirely in $S$, it has to coincide with $f$ in $S_1$, and take value $s_1$ in $S_3$. But the right limit of $\gamma$ at $0$ does not exist, since $f$ does not have a right limit at $0$. Therefore $\gamma$ cannot be continuous at $0$. In other words, there does not exist a continuous curve $\gamma$ joining $s_1$ and $s_2$.
\end{proof}


\item 14.D. Let $X=\left(x_n\right)$ be a sequence in $\boldsymbol{R}^p$ which is convergent to $x$. Show that $\lim \left(\left\|x_n\right\|\right)=\|x\| . \quad$ (Hint: use the Triangle Inequality.)
\begin{proof}
	We write $X^{(m)}$ for the sequence in $\mathbb{R}$ generated by the $m$-th coordinate of elements in  $X$. We write $x^{(m)}$ for the $m$-th coordinate of $x$.
	Since $X$ converges to $x$, we have, for each $m \in \{1, \ldots, p\} $, $X^{(m)}$ converges to $x^{(m)}$. Therefore for any $\varepsilon'>0$, there exists $N \in \mathbb{N}$ such that for all $n \ge N$, $|x_n^{(m)} - x^{(m)}| < \varepsilon'$.

	Now fix $\varepsilon>0$ and let $n$ be some natural number, to be adjusted later. By triangle inequality,
	\begin{align*}
		\|x_n - x\|
		&= \|\left( x_n^{(1)}-x^{(1)} \right) e_1 + \ldots + \left( x_n^{(p)} - x^{(p)} \right) \|e_p \\
		&\le \sum_{m=1}^{p} \|\left( x_n^{(m)} - x^{(m)} \right) e_m\|\\
		&= \sum_{m=1}^{p} \|\left( x_n^{(m)} - x^{(m)} \right)\| \|e_m\|\\
		&= \sum_{m=1}^{p} \|\left( x_n^{(m)} - x^{(m)} \right)\|\\
		&= \sum_{m=1}^{p} |\left( x_n^{(m)} - x^{(m)} \right)|
	.\end{align*}

For each $m$ in $1, \ldots, p$, let $\varepsilon' = \frac{\varepsilon}{p}$. Hence there exists $N_m$ such that for all $n > N_m$, $\left| x_n^{m} - x^{(m)} \right| < \varepsilon'$.

Now let $N = \max_p N_p$. For all $n\ge N$, and for any $m$, we have $n \ge N \ge N_m$, thus $|x_n^{(m)}- x^{(m)}|< \varepsilon' = \frac{\varepsilon}{p}$. Therefore
\begin{align*}
	\|x_n - x\|
		&= \sum_{m=1}^{p} |\left( x_n^{(m)} - x^{(m)} \right)|\\
		&\le  \sum_{m=1}^{p} \frac{\varepsilon}{p}\\
		&= \varepsilon
.\end{align*}

Since $\varepsilon$ is arbitrary and $N$ only depends on $\varepsilon$ and  $p$, we are done.
\end{proof}

\item 14.M. Let $X=\left(x_n\right)$ be a sequence of strictly positive real numbers such that $\lim \left(x_n^{1 / n}\right)<1$. Show that for some $r$ with $0<r<1$, then $0<x_n<r^n$ for all sufficiently large $n \in \mathbf{N}$. Use this to deduce that $\lim \left(x_n\right)=0$.
\begin{proof}
	Let $L = \lim_{n \to \infty} \left( x_n^{1 / n} \right) $. By assumpton, $0 \le L < 1$. Take $\varepsilon>0$ such that $L + \varepsilon<1$. Now there exists $N \in \mathbb{N}$ such that for all  $n \ge N$, we have $\left| x_n^{1 / n} - L \right| < \varepsilon$. Hence
	\[
		x_n < (\varepsilon + L)^n
	.\] 
	Let $r = \varepsilon + L$ and it is what we are looking for. Now $\lim_{n \to \infty} x_n \le \lim_{n \to \infty} r^n = 0$ since $0<r < 1$.
\end{proof}

\item 14.N. Let $X=\left(x_n\right)$ be a sequence of strictly positive real numbers such that $\lim \left(x_n^{1 / n}\right)>1$. Show that $X$ is not a bounded sequence and hence is not convergent.
\begin{proof}
	Let $L = \lim_{n \to \infty} \left( x_n^{1 / n} \right) $. By assumpton, $L > 1$. Take $\varepsilon>0$ such that $L - \varepsilon>1$. Now there exists $N \in \mathbb{N}$ such that for all  $n \ge N$, we have $\left| x_n^{1 / n} - L \right| < \varepsilon$. Hence
	\[
		x_n > ( L - \varepsilon)^n
	.\] 
	Let $r = L- \varepsilon$. For any $C > 1$, there exists a natural number $n$ such that $n > \frac{C - 1}{r - 1}$. Hence $1 + n (r-1) > C$. By Bernoulli's Inequality,  $r^n = (1 + (r-1))^n > C$. Thus $x_n > C$. This shows $x_n$ is unbounded, hence not convergent.
\end{proof}

\item 14.O. Give an example of a convergent sequence $\left(x_n\right)$ of strictly positive real numbers such that $\lim \left(x_n^{1 / n}\right)=1$. Give an example of a divergent sequence with this property.
\begin{proof}
	Convergent sequence: Let $x_n = 1$ for all $n$. Now $\lim_{n \to \infty} x_n = 1 = \lim_{n \to \infty} x_n^{1 / n}$.

	Divergent sequence: Let $x_n = n$ for all  $n$. Now we prove the convergnece of $x_n^{\frac{1}{n}}$. For all $n$ we have $n^{1 / n} > 1$.

	Let $a_n = x_n^{\frac{1}{n}} - 1 = n^{\frac{1}{n}}-1$.
	Now
	\[
		n = (n^{\frac{1}{n}})^n = (a_n + 1)^n
	.\] 
	By the Bernoulli Inequality,
	\[
		(a_n+1)^n > \frac{n(n-1)}{2}a_n^2
	\]
	
	Hence \[
		0 \le  a_n^2 \le  \frac{2n}{n(n-1)} \to 0 \quad \text{as } n \to 0
	.\] 
	By the squeeze theorem, $a_n^2 \to  0$, hence $a_n \to  0$. Hence $x_n^{\frac{1}{n}} = 1 + a_n \to 1$.
	

\end{proof}


\item 15.L. If $0<a \leq b$ and if $x_n=\left(a^n+b^n\right)^{1 / n}$, then $\lim \left(x_n\right)=b$.
\begin{proof}
	\begin{align*}
		\frac{x_n}{b} &= \frac{(a^n + b^n)^{1 / n}}{b}\\
		&=  \left( \frac{a^n + b^n}{b^n} \right) ^{1 / n}\\
		&= \left( \left( \frac{a}{b} \right) ^n+1 \right) ^{1 / n} \\
		&\le (1 + 1)^{n}\\
		& \to 1
	.\end{align*}
	To justify the inequality, note that since $0<a\le b$, we have $\frac{a}{b}\le 1$. Therefore $\left( \frac{a}{b} \right) ^n \le \frac{a}{b} \le 1$.

	To justify the final step, note that for all $c>1$ we have $\lim_{n \to \infty} c^{\frac{1}{n}} = 1$. Below is the proof:
\begin{lemma}
	For all $c > 1$, we have $\lim_{n \to \infty} c^{\frac{1}{n}} = 1$.

	\begin{proof}
		Since $x^k$ is an increasing function of $x$, for all positive $k$ and positive $x$, 
		\[
			c^{1 / n} > 1^{1 / n} = 1
		.\] 
		Fix $\varepsilon>0$. By Archimedean property, there exists $N \in \mathbb{N}$ such that $1 + N \varepsilon>c$, hence by Bernoulli Inequality, $(1 + \varepsilon)^N > c$. Therefore, $1+\varepsilon>c^{1 / N}$.

		Since $c^{1 / n}$ is a decreasing function of $n$, we have, for all $n \ge N$, \[
			c^{1 / n} \le c^{1 / N} < 1 + \varepsilon
		.\] 

		Hence for all $N \ge n$, \[
			1 <  c^{1 / n} \le  1 + \varepsilon 
		.\] 
		By squeeze theorem, $c^{1 / n} \to  1$.
	\end{proof}
	
\end{lemma}
	
\end{proof}

\item 15.M. Every irrational number in $\boldsymbol{R}$ is the limit of a sequence of rational numbers. Every rational number in $\boldsymbol{R}$ is the limit of a sequence of irrational numbers.
\begin{proof}
	We use the following two lemmas. They are already check in previous exercises.
\begin{lemma}
	$\mathbb{Q}$ is dense in $\mathbb{R}$. In other words, every open ball in $\mathbb{R}$ contains at least one element from $\mathbb{Q}$.
\end{lemma}

\begin{lemma}
	$\mathbb{R} \setminus \mathbb{Q}$ is dense in $\mathbb{R}$. In other words, every open ball in $\mathbb{R}$ contains at least one element from $\mathbb{R}\setminus \mathbb{Q}$.
\end{lemma}

Fix irrational $x$. For all $n \in \mathbb{N}$, use the first lemma to select $q_n \in \mathbb{Q} \cap B_{1 / n}(x)$. 
Now for all $\varepsilon>0$, there exists $N$ such that for all $n \ge N$, $\frac{1}{n} < \varepsilon$. Thus $|q_n - x| < \frac{1}{n} < \varepsilon$. Thus $q_n$ converges to $x$.


Fix rational $x$. For all $n \in \mathbb{N}$, use the second lemma to select $q_n \in \mathbb{R}\setminus \mathbb{Q} \cap B_{1 / n}(x)$. 
Now for all $\varepsilon>0$, there exists $N$ such that for all $n \ge N$, $\frac{1}{n} < \varepsilon$. Thus $|q_n - x| < \frac{1}{n} < \varepsilon$. Thus $q_n$ converges to $x$.
\end{proof}
