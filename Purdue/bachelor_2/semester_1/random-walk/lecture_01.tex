This document is in-progress work on a summer research project guided by Prof. Peterson, in collaboration with Zhe and Prof. Mountford.

The current research goal is to generalize the results in (\cite{kosyginaConvergenceNonconvergenceScaled2022}) about functional limit law for asymptotically free self-interacting random walk (AF-SIRW). (\cite{kosyginaConvergenceNonconvergenceScaled2022}) establishes convergence to BMPE in the case $p > \frac{1}{2}$. This research aims to generalize to the case  $p \le \frac{1}{2}$.

\section{Proof scheme of convergence of AF-SIRW to BMPE when $p < \frac{1}{2}$}

\subsection{Background and Definitions}

\subsubsection{Basic Notations}

\begin{enumerate}
	\item 
Let $X_n$ be the AF-SIRW of consideration. Denote the running maximum and minimum by $M_n$ and $I_n$.

\item Define $L_x(n) = \sum_{i = 0}^{n} \mathbf{1}(X_i = x)$ be the \textit{local time} of site $x$ at time $n$.

\item Define $\lambda_{x, m} = \inf \{n: L_x(n) \ge m\} $ be the time when the site $x$ gets the $m$-th visit.
Write $\mathcal{M}^{(x,m)}$ for $M_{\lambda_{x, m}}$.
	\item The \textit{total consumed drift} at time $n$ is
		\[
	C_n = \sum_{i=0}^{n-1} \mathbb{E}\left[ X_{i+1} - X_i | \mathcal{F}_i^X \right] 
		.\] 
where $\mathcal{F}_i^X = \sigma\left( X_k | k \le i \right) $.

	\item 
The \textit{local drift} at a point $x$ is
\[
\delta_{x} = \sum_{i=0}^{\infty} \mathbb{E}\left[ X_{i+1} - X_i | \mathcal{F}_i^X \right] 
\mathbf{1}\left( X_i = x \right) 
.\] 

\item
The \textit{local drift by $m$-th visit} at a point $x$ is
\[
\delta_{x,m} = \sum_{i=0}^{\lambda_{x, m}} \mathbb{E}\left[ X_{i+1} - X_i | \mathcal{F}_i^X \right] 
\mathbf{1}\left( X_i = x \right) 
.\]
where $\lambda_{x,m}$ is the stopping time when $x$ has been visited $m$ times.

\item The local drift at site $y$ at time $\lambda_{x, m}$ is written as
	\[
		\Delta_y^{(x,m)} = \sum_{i=0}^{\lambda_{x, m}} \mathbb{E}\left[ X_{i+1} - X_i | \mathcal{F}_i^X \right] \mathbf{1}(X_i = y)
	.\] 

\end{enumerate}
\subsubsection{Main scheme}


 To show convergence to $(\gamma, \gamma)$-BMPE, we first verify that $B_n = X_n - C_n$ is a martingale with respect to the filtration $\mathcal{F}_n$, and show that it converges to a Brownian motion.

The next step is to control the accumulated drift:

\begin{lemma}
	\label{lem:drift-term}
	For any $t \ge 0$ and any $\varepsilon>0$,
	\[
		\lim_{n \to \infty } P\left( 
		\sup_{k \le n t}
		\left| C_k + \gamma(M_k) + \gamma(I_k) \right| 
		\ge \varepsilon \sqrt{n} 
	\right)  =0
	.\] 
\end{lemma}
For reference, see (Lemma 4.2, \cite{kosyginaFunctionalLimitLaws2016}).

Write
\begin{align}
	C_n^+ &= \sum_{y > X_n} \sum_{i = 0}^{n-1} \mathbb{E}\left[ X_{i + 1} - X_i | \mathcal{F}_i^X \right] \mathbf{1}(X_i = y)\\
	C_n^0 &= \sum_{i = 0}^{n-1} \mathbb{E}\left[ X_{i + 1} - X_i | \mathcal{F}_i^X \right] \mathbf{1}(X_i = y) \\
	C_n^- &= \sum_{y < X_n} \sum_{i = 0}^{n-1} \mathbb{E}\left[ X_{i + 1} - X_i | \mathcal{F}_i^X \right] \mathbf{1}(X_i = y)
.\end{align}

$C_n^0$ should be negligible in the argument.
We only discuss the first claim about $C_n^+$, and $C_n^-$ should follow by symmetry.


	Loosely speaking, we want the following estimate (at most places) for local drifts
\begin{equation}
	\sum_{i = 0}^{n-1} \mathbb{E}\left[ X_{i + 1} - X_i | \mathcal{F}_i^X \right] \mathbf{1}(X_i = y)
	\approx \text{sgn}(y) \gamma
.\end{equation} 
which amounts to showing
\begin{equation}
	\label{eqn:approx-goal}
	C_n^+ \approx \sum_{y > X_n} \text{sgn}(y)\gamma = \begin{cases}
		\gamma(M_n - X_n) & X_n \ge 0\\
		\gamma(M_n + X_n) & X_n < 0\\
	\end{cases}
.\end{equation}



\subsubsection{Restrict to excursion times}


	We try to work on events $\{n = \lambda_{x, m}\}  = \{X_n = x, L_x(n) = m\} $ and establish \eqref{eqn:approx-goal}. Under this event we have
	\[
		C_n^+ = \sum_{y = x+1}^{\mathcal{M}^{(x,m)}} \Delta_y^{(x,m)}
	.\] 

	For simplicity of discussion, consider $y > x > 0$. Then our goal becomes showing
	\begin{equation}
		\label{eqn:goal-2}
		\sum_{y = x+1}^{\mathcal{M}^{(x,m)}} \left(\Delta_y^{(x,m)} - \gamma\right) = o(\sqrt{n} )
	.\end{equation} 

\subsubsection{BLP construction}

Let $\mathcal{E}_{y-1}^{(x,m)}$ denote the number of upcrossings from $y-1$ to $y$ by the time $\lambda_{x, m}$.
	From now on we write $\mathcal{E}$ in place of $\mathcal{E}_{y-1}^{(x,m)}$ whenever there is no ambiguity. 
	
	We know that $\left( \mathcal{E}_y^{(x,m)} \right) _{y \ge  x}$ is a branching-like process (BLP). From analysis of BLP, $\Delta_y^{(x,m)} \in \sigma\left( \mathcal{E}_{y-1}^{(x,m)}, \mathcal{E}_y^{(x,m)} \right) $. 
	Write $\mathcal{G}_{z}^{(x,m)} := \sigma\left( \mathcal{E}_y^{(x,m)}: x \le y \le z \right) $, and note that $\left( \Delta_y^{(x,m)} \right) _{y \ge  x}$ is adapted to $\left( \mathcal{G}_y^{(x,m)} \right) _{y \ge  x}$.
	Let
 \begin{equation*}
	 \rho_y^{(x,m)} = \mathbb{E}\left[ \Delta_y^{(x,m)} | \mathcal{G}_{y-1} ^{(x,m)} \right] 
.\end{equation*} 
and we can decompose \eqref{eqn:goal-2} to a martingale term and a residue term:
\[
		\sum_{y = x+1}^{\mathcal{M}^{(x,m)}} \left(\Delta_y^{(x,m)} - \gamma\right) = \sum_y \left(\Delta_y^{(x,m)} - \rho_y^{(x,m)}\right) + \sum_y \left(\rho_y^{(x,m)} - \gamma\right)
.\] 

The rest of this document discusses how we can bound
\begin{equation}
	\label{eqn:msum-goal-1}
	\sum_{y = x+1}^{\mathcal{M}_{x,m}} \Delta_y^{(x,m)} - \rho_y^{(x,m)} = o(\sqrt{n}) 
.\end{equation}
