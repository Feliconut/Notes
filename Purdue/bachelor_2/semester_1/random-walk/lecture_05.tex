\subsection{Local drift control without moment bound}

Assume that $y > x >  0$, let $m > 0$. We want to show \eqref{eqn:msum-goal-1}.
We work on the event $\left\{ X_n = x, n = \lambda_{x, m} \right\} $. Consider some $K>0$ and condition further on the events
\begin{align}
	\mathcal{M}^{(x,m)} &< K \sqrt{ n} \label{eqn:good-event-1}\\
	\mathcal{E}_y^{(x,m)} &< K \sqrt{n} \quad &\forall y>x\\
	\left| \tau_i^{\mathcal{B}} - 2 i \right| &< \sqrt{ i} \log n \quad &\forall i < K \sqrt{n} \\
	\left| \tau_{i+1}^{\mathcal{B}} - \tau_i^{\mathcal{B}} \right| &< \log n \quad &\forall i < K \sqrt{n} \label{eqn:good-event-2}
.\end{align} 

We denote the four events above as the \textit{good event} $G$.
We want a deterministic bound on local drift at $y$ consumed up until $K\sqrt{n}$-th left step. This will bound $\Delta_y^{(x,m)}$ on the good event.

\begin{equation*}
	\left| \Delta_y^{(x,m)} \right| 
	= 
	\left| 	\sum_{i = 0}^{\mathcal{E}_y^{(x,m)}} 
	\sum_{l = \tau_i^{\mathcal{B}}} ^{\tau_{i+1}^{\mathcal{B}}  -1}
	\alpha(\mathcal{B}_l, \mathcal{R}_l)
	\right| 
	\le 
	\sum_{i = 0}^{K \sqrt{n} } 
	\sum_{l = \tau_i^{\mathcal{B}}} ^{\tau_{i+1}^{\mathcal{B}}  -1}
	\left|
	\alpha(\mathcal{B}_l, \mathcal{R}_l)
	\right| 
	.
\end{equation*}

\begin{align*}
	&\sum_{i = 0}^{K \sqrt{n} } 
	\sum_{l = \tau_i^{\mathcal{B}}} ^{\tau_{i+1}^{\mathcal{B}}  -1}
	|\alpha(\mathcal{B}_l, \mathcal{R}_l)|
	:=
	\sum_{i = 0}^{K \sqrt{n} } 
	\sum_{l = \tau_i^{\mathcal{B}}} ^{\tau_{i+1}^{\mathcal{B}}  -1}
	\left|\frac{-w(2 \mathcal{B}_l + 1) + w(2 \mathcal{R}_l)}{w(2\mathcal{B}_l + 1) + w(2 \mathcal{R}_l)}\right|\\
	&\le C_w \sum_i \sum_l \left| \frac{1}{w(2 \mathcal{R}_l)} - \frac{1}{w(2 \mathcal{B}_l + 1)} \right|  \\
	&= C_w \sum_i \sum_l \left| \frac{1}{w(2 l - 2 i)} - \frac{1}{w(2i + 1)} \right|  \\
	&= C_w \sum_i \sum_{j = \tau_i^{\mathcal{B}} - i}^{\tau_{i+1}^{\mathcal{B}} - i - 1} \left| \frac{1}{w(2j)} - \frac{1}{w(2i + 1)} \right|  \\
	&= C_w \sum_i \sum_{j = \tau_i^{\mathcal{B}} - i}^{\tau_{i+1}^{\mathcal{B}} - i - 1} \left|  2^p B\left( \frac{1}{(2j)^p} - \frac{1}{(2i + 1)^p} \right)  + O\left( \frac{1}{i^{\kappa + 1}} \right) \right|  \\
	&\le C_w \sum_i \log n \sup_{|j - i| \le 2 \sqrt{i}  \log n} \left|  2^p B\left( \frac{1}{(2j)^p} - \frac{1}{(2i + 1)^p} \right)  + O\left( \frac{1}{i^{\kappa + 1}} \right) \right| \\
	&\le C_{w, p} \sum_i \log n \left( 
		(4 \sqrt{ i } \log n) (2 i - 2 \sqrt{ i } \log n)^{- p - 1} + (2 i)^{- p - 1} + O(i^{- \kappa - 1})
	\right)  \\
	&\le C_{w, p} \sum_i \log n \left( i ^{-p - \frac{1}{2}} \log n +  i^{- \kappa - 1} \right)  \\
	&= C_{w, p} \sum_i i^{- p - \frac{1}{2}} \log^2 n + i^{- \kappa - 1 } \log n \\
	&\stackrel{(p<\frac{1}{2})}{\le } C_{w, p} \left( (K \sqrt{ n} )^{-p+ \frac{1}{2}} \log^2 n + (K \sqrt{ n} )^{- \kappa } \log n \right)  \\
	&= C_{w, p, K} n^{-\frac{1}{2}p + \frac{1}{4}  }  \log^2 n \qquad \text{assuming $\kappa$ small}
.\end{align*} 

We assume it has been proved that $\rho_y^{(x,m)} \to \gamma$ at rate $o(n^{-\frac{1}{2}})$ uniformly in $y$. Then with probability going to $1$ as $n\to \infty $, $\left| \Delta_y^{(x,m)} - \rho_y^{(x,m)} \right| < C n^{-\frac{1}{2}p + \frac{1}{4}  }  \log^2 n$ for all $y > x$. Denote this ``Lipchitz event'' as $F$, and have $P(F^c) \to 0$.

Then we can apply the Azuma's inequality on $G \cap F$:

\begin{align}
	P\left( \sum_y (\Delta_y^{(x,m)} - \rho_y^{(x,m)}) > \varepsilon \sqrt{n}  \right) \notag
	&\le \exp\left( - \frac{\varepsilon^2 n}{2 K \sqrt{n} \left( C_{w, p, K} n^{-\frac{1}{2}p + \frac{1}{4}} \log^2 n  \right)^2 } \right) + P(G^c) + P(F^c) \notag \\
	&= \exp\left( - C_{w, p, K, \varepsilon} \, n^{p } \log^{-4} n \right) +P(G^c) + P(F^c) \label{eqn:drift-martingale}
.\end{align}

\begin{remark}
	The bound above is under assumption that  $p < \frac{1}{2}$. For $p = \frac{1}{2}$, we get
	\begin{multline}
		\sum_{i = 0}^{K \sqrt{n} } 
		\sum_{l = \tau_i^{\mathcal{B}}} ^{\tau_{i+1}^{\mathcal{B}}  -1}
		\left|\frac{-w(2 \mathcal{B}_l + 1) + w(2 \mathcal{R}_l)}{w(w\mathcal{B}_l + 1) + w(2 \mathcal{R}_l)}\right| 
	\stackrel{(p=\frac{1}{2})}{\le } C_{w, p, K} \left(
	\log^3 n + (K \sqrt{ n} )^{- \kappa } \log n \right)  
	\\
	\le C_{w, p, K} \, n^{- \frac{\kappa}{2}} \log n
	\end{multline} 
	which yields the bound
\begin{align*}
	P\left( \sum_y (\Delta_y^{(x,m)} - \rho_y^{(x,m)}) > \varepsilon \sqrt{n}  \right) 
	&\le \exp\left( - \frac{\varepsilon^2 n}{2 K \sqrt{n} \left( C_{w, p, K} n^{-\frac{\kappa}{2}} \log n  \right)^2 } \right) + P(G^c) + P(F^c)\\
	&= \exp\left( - C_{w, p, K, \varepsilon} \, n^{\frac{1}{2} + \kappa } \log^{-2} n \right) +P(G^c) + P(F^c)
.\end{align*}
\end{remark}

\begin{remark}
	The proof above works under the weaker assumption that  $\left| \tau_{i+1}^{\mathcal{B}} - \tau_i^{\mathcal{B}} \right| < i^{\kappa'} \log n \quad \forall i < K \sqrt{n} 
.$
Under this assumption, the final bound we get ($p<\frac{1}{2}$) is 
\[
	P\left( \sum_y (\Delta_y^{(x,m)} - \rho_y^{(x,m)}) > \varepsilon \sqrt{n}  \right) 
	\le \exp\left( - C_{w, p, K, \varepsilon} \, n^{p - 2 \kappa'} \log^{-4} n \right) +P(G^c) + P(F^c)
\] 
	which still vanishes when $\kappa' < \frac{p}{2}$.
\end{remark}
