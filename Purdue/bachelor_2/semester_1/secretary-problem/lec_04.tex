\section{Calculation with error involved}

The weights $X_1, \ldots, X_n \sim \text{Unif}(0,1).$
The noisy observations $\tilde{X_1},\ldots,\tilde{X_n} \sim \text{Unif}(x_i-\delta, x_{i}+\delta).$

The strategy is to introduce thresholding parameters $R$ and $T \in (0,1).$ We examine the upcoming item $i$ if $\tilde{x}_{i}>R$, and accept the item if $x_{i}>T$.
We still want to find $\prob{X_{max} \text{ is selected}}$ given particular $R,T,n,\delta$.

\begin{equation}
	\begin{split}
	    \prob{X_{max} \text{ selected}}&=\sum_{i=1}^{n-2} \int_0^1 \prob{X_i \text{best} \cap i \text{ selected} | X_{i} = x} f(x) \, dx \\
	    &\qquad + \prob{\text{reach }n-1 \cap X_{n-1} \text{ best} \cap \tilde X_{n-1} > R} \\
	    &\qquad + \prob{\text{reach }n \cap X_n \text{ best}}
	\end{split}
\end{equation}

\begin{align*}
    \intertext{For all $i\le i-2$:}
	&\prob{X_i \text{best} \cap i \text{ selected} | X_{i} = x} \\
	&= \prob{\text{reach }i \cap \tilde{X_{i}}>R \cap X_{i}>T \cap X_i \text{ max} | X_i= x} \\
	&= \prob{\text{Unif}(x-\delta, x+\delta)>R}\ind{\left( x>T \right) }\prob{\text{reach }i \cap X_i \max | X_i = x}
\end{align*}
	
\begin{align*}
    \intertext{For all $i\le i-2$:}
	&\prob{\text{reach }i \cap X_i \max | X_i = x} \\
	&= \sum_{k=0}^{\left\lfloor \frac{i-1}{2} \right\rfloor}
	\prob{\text{reach }i \cap X_i \max \cap k \text{ examines before reaching }i | X_i = x}\\
	&= \sum_{k=0}^{\left\lfloor \frac{i-1}{2} \right\rfloor}
	\prob{\text{reach $i$ with $k$ examines}}\prob{X_i \max |\text{reach $i$ with $k$ examines} \cap X_i = x} \\
	&= \sum_{k=0}^{\left\lfloor \frac{i-1}{2} \right\rfloor}
	\underbrace{ \frac{\left( i-1-k \right) !}{k!\left( i-1-2k \right) !} }_{\text{ ways to reach $i$ }}
	\prob{\text{ED}}^k \prob{\text{D}}^{i-1-2k} 
	\underbrace{ F(x | X<T)^k  }_{\text{examined}}
	\underbrace{ F(x)^k  }_{\text{omitted}}
	\underbrace{ F(x | \tilde X < R)^{i-1-2k}  }_{\text{declined directly}}
	\underbrace{ F(x)^{n-i} }_{\text{upcoming}}
\end{align*}
where
\begin{align*}
	&\prob{\text{ED}} &=&\frac{1}{2\delta} \int_0^T \int_{y-\delta}^{y+\delta} \ind(z>R) \,dz \,dy = \text{const}
	\\
	&\prob{D} &=&\frac{1}{2\delta} \int_0^1 \int_{y-\delta}^{y+\delta} \ind(z<R) \,dz \,dy = \text{const} 
	\intertext{and for $0\le x\le 1$:}
	&F(x|X<T) &= &\begin{cases}
		\frac{x}{T}	&0\le x<T \\
		1		&x\ge T
	\end{cases}
	\\
	&F(x | \tilde X < R) & = & \frac{\int_0^x \int_{y-\delta}^{y+\delta} \ind(z<R) \,dz \,dy}{\int_0^1 \int_{y-\delta}^{y+\delta} \ind(z<R) \,dz \,dy}
	\\
	&F(x) & = & x
.\end{align*}

\section{Another Approach to Estimate $P^{\max}$}
To simplify the writing, we write $\text{Error Win}$ as the event that we find the best secretary under the model with error. We write $\text{Original Win}$ as the event that we find the best secretary under the original model. 

\begin{align*}
    \Pr(\text{Error Win}) &= \Pr(\text{Error Win} \vert \text{Original Win}) \Pr(\text{Original Win}) \\&+ \Pr(\text{Error Win} \vert \text{Original Lose}) \Pr(\text{Original Lose}) \\& \geq 0.485 \Pr(\text{Error Win} \vert \text{Original Win}).
\end{align*}

{\color{red} Optional exercise: Is there any method to bound $\Pr(\text{Error Win} \vert \text{Original Lose})$. }

Next, we bound the probability of $\text{Error win}$ conditional on $\text{Original Win}$. We start with defining two events: we let $\mathcal E_x=\max X_i = x$. We let $\mathcal F_y$ be that the secretary who arrives in the last period before the max secretary arrives. Because the arriving process is i.i.d, it is easy to see that $\mathcal F_y \sim \text{Unif}(0,1)$.

{\color{red} Exercise1: Find the distribution of $\mathcal E_x$.}

{\color{blue} Hint: See Beta distribution.}

We have 
\begin{align*}
    \Pr(\text{Error Win} \vert \text{Original Win}) &= \int_{x,y}  \Pr(\text{Error Win} \vert \text{Original Win}, \mathcal E_x, \mathcal F_y) f(x) dxdy.
\end{align*}

We construct two arriving instances as follows: let $I_1$ contains $n$ i.i.d. $\text{Unif}(0,1)$ r.v. In additional, the maximal realization is $x$. The last r.r. before the max one has realization $y$. Let $I_2$ be a copy of $I_1$. Conditional on $\text{Original win}$, there are two cases that we do not win on the same instance:

\begin{itemize}
    \item The max $x$ has observation $\hat{x} < T$, then we do not examine the best secretary.
    \item The one before max, $y$, has observation $\hat{y} \geq T$, then the best secretary is skipped. (Notice that here we do not need to consider the second last before $x$ and all the previous r.v., because we are providing the lower bound.)
\end{itemize}

Therefore, we have
\[
\Pr(\text{Error lose} \vert \text{Original Win}, \mathcal E_x, \mathcal F_y) \leq \Pr(\text{Case 1} \cup \text{Case 2}) \leq \Pr(\text{Case 1})+\Pr(\text{Case 2}).
\]

{\color{red} Exercise 2: Find the probability of Case 1 and Case 2 by considering that the observation cannot exceed $1$ or below $0$.}

All calculation were based on the assumption that $\delta < T < 1-\delta$.

\begin{gather*}
    F(x) = \begin{cases}
    1 & x<T-\delta \\
    0 & x>T+\delta \\
    \frac{x+\delta-T}{2\delta} & \text{otherwise}
    \end{cases} \\
    \Pr(\text{Case 1}) = F(x)\\
    \Pr(\text{Case 2}) = 1-F(y)
\end{gather*}

Finally, we have 
\begin{align*}
    \Pr(\text{Error Win}) &\geq  0.485 \Pr(\text{Error Win} \vert \text{Original Win}) \\&\geq 0.485 \int_{x,y} (1-\Pr(\text{Case 1})-\Pr(\text{Case 2}))f(x)dxdy.
\end{align*}
