\subsubsection{
	Response to \eqref{ex1}
}
\begin{proof}
	
Use induction on $n$. The base case is trivial: \[
1 = \frac{1}{k+1}+ O(1)
.\] 
Now suppose the property holds for $n$.
\begin{align*}
	\sum_{i=1}^{n+1} {i^k}&= (n+1)^k + \sum_{i=1}^{n} {i^k} = (n+1)^k + \frac{1}{k+1}n^{k+1} + O(n^k) \\
	\frac{1}{k+1}(n+1)^{k+1} &= \frac{1}{k+1}n^{k+1} + O(n^k)
.\end{align*}
By considering $(n+1)^k = O(n^k)$ and $O(n^k) \pm O(n^k) = O(n^k)$, the difference of the right hand side of the above equations is $O(n^k)$. The left hand side shows the property for $n+1$.
\end{proof}

\subsubsection{
	Response to \eqref{ex2}
}
Optimal P for a determininstic algorithm is $0.75$.
\begin{proof}
	$\text{ALG}$ is competely characterized by a function $\alpha: [0,1] \to [ 0,1 ]$, where $\alpha(x)$ represents the probability of accepting  $X_1$ after an observation $X_1 = x$.

\begin{align*}
	\operatorname{Pr}\left[{\text{win}}\right] 
	&= \int_0^1 \alpha \operatorname{Pr}\left[{X_2<x}\right] + (1-\alpha)\operatorname{Pr}\left[{X_2>x}\right] f(x) \, dx \\
	&= \int_0^1 \alpha x + (1-\alpha)(1-x) \,dx\\
	&= \int_0^1 2 \alpha x - \alpha \,dx + \frac{1}{2}\\
	&\le  \max_{\alpha} \int_0^1 2\alpha x - \alpha \,dx + \frac{1}{2}
.\end{align*}
$\text{ALG}$ is deterministic iff $\alpha(x) \in \{0,1\}$. If $\alpha$ takes $0$, the integral is zero. If $\alpha$ takes $1$, $2\alpha x - a > 0$ iff $x > \frac{1}{2}$. Hence to maximize the integral we take $\alpha(x) = \begin{cases}
	1 &x\ge \frac{1}{2}\\
	0 &x<\frac{1}{2}
\end{cases}$, which gives an upper bound of $\frac{3}{4}$.
\end{proof}
