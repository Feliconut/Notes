\subsection{Finding Image of Functions}

\begin{example}
	Find the image of vertical line $x=1$ under $f(z) = z^2$.
\begin{enumerate}
	\item $z(t) = 1 + it, -\infty <t<\infty $
	\item Plug \[
	w(t) = z^2 (t) = (1+it)^2 = 1-t^2 + 2 it
	.\] 
	So $u(t) = 1-t^2, v(t) = 2t$
\item Eliminate  $t$. $t = \frac{v}{2}$, u = $1-\frac{v^2}{4}$. This is a parabola.
\end{enumerate}
\end{example}
\begin{example}
	Find the preimage of vertical line $\operatorname{Re} w = 1$ under $f(z) = z^2$.
	\begin{align*}
		w &=  u + iv, \operatorname{Re} w = 1 \\
		z &=  x + iy \\
		w &=  (x+iy) \\
		&= x^2 - y^2 + 2 i xy \\
		\operatorname{Re}  w &= x^2 - y^2\\
		\operatorname{Im}  w &=  2x y
	\end{align*}
	This is a hyperbola.
	
\end{example}

\begin{example}
	The \logseq{Joukowski function} is defined as
	\[
	J(z) = \frac{1}{2}(z + \frac{1}{z})
	.\]
	\begin{enumerate}
		\item Find a 1-1 domain. The full domain is $\mathbb{C}^* = \{z \in \mathbb{C}: z \neq  0\} $.
	To decide where is it $1-1$, let
	\begin{align*}
		& J(z_1) = J(z_2)\\
		&\implies z_1-z_2 - \frac{z_1-z_2}{z_1z_2} = 0 
	.\end{align*}
	If $z_1 \neq  z_2$ then $z_1z_2 = 1$. We want to find a region where $z_1z_2 \neq  1$ for any $z_1z_2$.

	We can take the region inside unit circle. Consider the region $\mathcal{D}_1 = \{z: |z|<1\} $.
	We proceed to visualize its image. Fill it with circular curves.

\item Paramaterize the curve: $z = r_0 e^{it}$, $0\le t\le 2\pi$, where $r_0 \in (0,1)$. 
\item Plug into function $J$: \[
		w = \frac{1}{2}\left( r_0 e^{it }+r_0^{-1}e^{-it} \right) 
		= \frac{1}{2}\left( r_0 + \frac{1}{r_0} \right) \cos(t) + i \frac{1}{2} \left(r_0 - \frac{1}{r_0}\right) \sin(t).
	\]
\item Eliminate $t$:
	\[
	\frac{u^2}{\left(\frac{1}{2} \left( r_0 + \frac{1}{r_0} \right)  \right) ^2} 
	+ \frac{v^2}{\left(\frac{1}{2} \left( r_0 - \frac{1}{r_0} \right)  \right) ^2} = 1
	.\] 

	The image of a circle of radius $r_0$ centered at the origin is an ellipse with half-axis $\frac{1}{2} \left( r_0 + \frac{1}{r_0} \right)   $ and $\frac{1}{2} \left( r_0 - \frac{1}{r_0} \right)  $.

	Let $r_0$ continuously change from $0$ to $1$. The first half-axis decreases from $+\infty $ to $1$ ; The second half-axis decreases from $+\infty $ to $0$. 

	Conclusion: the image of $|z| < 1$ is the plane with cut from $-1, 1$.


	\end{enumerate}
Other properties of $J$ :
\begin{itemize}
\item $J$ is conformal except at two points, $-1$ and $1$. The angle of $\pi$ is expanded to $2 \pi$ at these two points.
\item $J$ is 1-1 in upper and lower unit semi-disks. It maps the upper unit semi-disk to the lower half plane, and the lower unit semi-disk to the upper half plane.
\item $J$ maps the upper half plane with unit disk removed to the upper half plane, and similarly for lower half plane.
\item $J(\pm i) = 0, J(0) = \infty$, $J(-1) = -1, J(1) = 1$.
\item $J$ is 1-1 also in $\operatorname{Im} z > 0$ and $\operatorname{Im}  z < 0$.
\end{itemize}
\end{example}
