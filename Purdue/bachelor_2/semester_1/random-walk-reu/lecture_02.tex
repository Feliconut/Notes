\subsection{Bounding martingale part using variance bound}

By Markov property of $\Delta_y^{(x,m)}$, we have $\rho_y \in \sigma\left( \mathcal{E}_{y-1}^{(x,m)} \right) $. We claim that

\[
	\operatorname{Var} \left[ \Delta_y^{(x,m)} - \rho_y^{(x,m)} | \mathcal{E} \right] = O(\mathcal{E}^{1- 2p})
.\] 

From this we obtain a variance tail bound
\[
	P\left( \left| \Delta_y^{(x,m)} - \rho_y^{(x,m)} \right| > \varepsilon n^{\frac{1}{4} - \frac{\kappa}{2}}   | \mathcal{E}_{y - 1} \right)
	\le  \frac{C \mathcal{E}^{1 - 2 p}}{\varepsilon^2 n^{\frac{1}{2} - \kappa}} 
\]

Now restrict our attention to a good event where all the $\mathcal{E}_y^{(x,m)}$ are bounded by $C \sqrt{n}$ uniformly in $y$, a event with probability arbitrarily close to 1. Then taking expectation yields
\[
	P\left( \left| \Delta_y^{(x,m)} - \rho_y^{(x,m)} \right| > \varepsilon n^{\frac{1}{4} - \frac{\kappa}{2}}  \right)
	\le  \frac{C n^{\frac{1}{2} - p}}{\varepsilon^2 n^{\frac{1}{2} - \kappa}} = O(n^{-p + \kappa})
\]

When $p > \frac{1}{2}$, this shows the martingale sum is almost Lipchitz, allowing application of Azuma inequality. Let $\kappa \in (0,p - \frac{1}{2})$,

\[
	\sum_y \operatorname{Pr} \left[ \left| \Delta_y^{(x,m)} - \rho_y^{(x,m)} \right| >  \varepsilon n^{\frac{1}{4} - \frac{\kappa}{2}} \right]
	= O\left( n^{\frac{1}{2} - p + \kappa} \right) 
	=: \eta
.\] 

\begin{multline}
	\operatorname{Pr} \left[ \sum_y  \left| \Delta_y^{(x,m)} - \rho_y^{(x,m)} \right| >  \varepsilon \sqrt{n}  \right]
	\le \eta + 
	2 \exp\left( - \frac{\varepsilon^2 n}{ 2 \sqrt{n} (\varepsilon n^{\frac{1}{4} - \frac{\kappa}{2}})^2 } \right) \\
	= O\left( n^{\frac{1}{2} - p + \kappa} \right) + O\left( \exp\left( - \frac{n^\kappa}{2} \right)  \right)   
	= O\left( n^{\frac{1}{2} - p + \kappa} \right)
	\label{eqn:azuma-bound-1}
.\end{multline} 

When $p >  \frac{1}{2}$ this is giving something that goes to zero. We need $o(n^{-1})$ bound, which is necessary for passing from stopping times $\lambda_{x,m}$ to general times $n$, and this happens only for $p > \frac{3}{2}$. It turns out that a higher moment tail bound is needed.

\subsection{Controlling variance of local drift}

\subsubsection{Background and Notation}

The generation of steps at one fixed site can be described by the generalized Polya Urn model.

\begin{notation}
	$\mathcal{B}$ refers to blue ball or left step; $\mathcal{R}$ refers to red ball or right step.
\end{notation}


\begin{notation}
	For simplicity of notation, define $\alpha(i,j) := \frac{- l(i) + r(j)}{l(i) + r(j)} = \frac{-w(2 i - 1) + w(2j)}{w(2i-1)+w(2j)}$. This is the encountered drift at some node when the Polya Urn state is $(i,j)$.
\end{notation}

Also, $\mathcal{E}$ refers to $\mathcal{E}_y^{(x,m)}$, unless otherwise specified.

We have the following branching process representation of $\rho_y^{(x,m)}$, where the inner sum represents the ``descendants'' produced by the $i$-th ``ancestor'', i.e. the number of right excursions made from $y$ between the $i -1$-th and $i$-th left excursion.
\[
	\rho_y^{(x,m)} = \mathbb{E}\left[ \sum_{i = 1}^{\mathcal{E}_{y-1}^{(x,m)}} 
		\sum_{j = \tau_{i-1}^{\mathcal{B}}-i+1}^{\tau_i^{\mathcal{B}}-i}   
		\alpha(i, j) 
\right] 
	.
\]

\subsubsection{Estimation of variance}
 
First compute variance by conditioning. By martingale property, the second term in conditional variance vanishes:
 \begin{align*}
	  &\operatorname{Var} \left[ \Delta_y^{(x,m)} - \rho_y^{(x,m)}  \right]\\
	 &=  \mathbb{E} \left[\operatorname{Var} \left[ \Delta_y^{(x,m)} - \rho_y^{(x,m)} | \mathcal{E}_{y-1}^{(x,m) } \right]\right]
	 +  \operatorname{Var}\left[ \mathbb{E} \left[ \Delta_y^{(x,m)} - \rho_y^{(x,m)} | \mathcal{E}_{y-1}^{(x,m) } \right]\right]\\
	 &=  \mathbb{E} \left[\operatorname{Var} \left[ \Delta_y^{(x,m)} | \mathcal{E}_{y-1}^{(x,m) } \right]\right]\\
	 &= \mathbb{E}\left[ \operatorname{Var} \left[ 
	 \sum_{i = 1}^\mathcal{E} 
		\sum_{j = \tau_{i-1}^{\mathcal{B}}-i+1}^{\tau_i^{\mathcal{B}}-i}   
		\alpha(i, j) 
	 \bigg| \mathcal{E}_{y-1}^{(x,m)}
\right]  \right] 
.\end{align*}

We claim that
\begin{equation}
	\label{eqn:var-func}
	\varphi(w):=\operatorname{Var} \left[ \sum_{i = 1}^w 
		\sum_{j = \tau_{i-1}^{\mathcal{B}}-i+1}^{\tau_i^{\mathcal{B}}-i}   
		\alpha(i, j) 
\right] 
	= O(w^{1 - 2 p})
.\end{equation} 

Thus, 
\[
	  \operatorname{Var} \left[ \Delta_y^{(x,m)} - \rho_y^{(x,m)}  | \mathcal{E} \right]
	  = O(\mathcal{E}^{1 - 2 p})
.\] 

Now justify \eqref{eqn:var-func}.
Split the first sum into sum of covariances; apply Hölder inequality to control all terms by the diagonal. Then it suffices to justify
\[
	\operatorname{Var} \left[
		\sum_{j = \tau_{i-1}^{\mathcal{B}}-i+1}^{\tau_i^{\mathcal{B}}-i}   
		\alpha(i, j) 
\right] 
	= O\left( i^{-1-2p} \right) 
.\] 

Asymptotically all the $\tau_i - \tau_{i-1}$ converge in distribution to $\text{Geom}(\frac{1}{2})$, so they can be treated as $O(1)$. Then we can apply the result of lemma~\ref{lemma:variance-diagonal} to obtain the claim above. (Also, this claim about diagonal terms has been verified by simulation.)



\subsubsection{Moment Estimates for variance bound}


\textbf{Notations.} For simplicity of notation, define $\alpha(i,j) := \frac{- l(i) + r(j)}{l(i) + r(j)} = \frac{-w(2 i - 1) + w(2j)}{w(2i-1)+w(2j)}$. This is the encountered drift at some node when the Polya Urn state is $(i,j)$.


We have the branching process representation of $\rho_y^{(x,m)}$ :
\[
	\rho_y^{(x,m)} = \mathbb{E}\left[ 
	\sum_{i = 1}^{\mathcal{E}_{y-1}^{(x,m)}} 
		\sum_{j = \tau_{i-1}^{\mathcal{B}}-i+1}^{\tau_i^{\mathcal{B}}-i}   
		\alpha(i, j) 
	\right] 
.\] 


\begin{lemma}
(Lemma 4.1 and Remark 4.2, \cite{kosyginaFunctionalLimitLaws2016})
	\[
		P\left( \left| \tau_i^{\mathcal{B}} - 2i\right| \ge m \right)  \le C e ^{\frac{-c m^2}{ m \vee n}}
	.\] 
	\label{lem:section-concentration}
\end{lemma}

\begin{lemma}
	\[
		\mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right) \right] \le O(i^{-p-\frac{1}{2}})
	.\] 
	\label{lem:sectional-drift-1}
\end{lemma}
\begin{proof}
	Calculate by splitting into good and bad events. Define good event $G = \{|\tau_i^{\mathcal{B}} - 2i| < m \sqrt{ i} \} $. 

\begin{align*}
	\mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right) \right]
	&= \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right) \mathbf{1}(G)\right] + \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right) \mathbf{1}(G^c)\right] \\
	&\le \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right)  | G\right] + P(G^c) \\
	&= \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right)  | G\right] + O(e^{-c m^2}) 
.\end{align*}

\begin{align*}
	 \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right)  | G\right]
	 &\le \sup _{j:|i - j| < m \sqrt{i} } \alpha(i,j) \\
	 &\le C \sup \left| 2^p B \left( \frac{1}{(2i - 1)^p}  - \frac{1}{(2j)^p} + O\left(\frac{1}{i^{\varkappa
	 + 1}}\right) \right) \right|  \\
	 &\le O\left(\frac{1}{i^{\varkappa + 1}}\right) 
	 + C \sup \left| \frac{1}{i^p} - \frac{1}{j^p}\right| + C \sup \left| \frac{1}{(2i-1)^p} - \frac{1}{(2i)^p} \right| \\
	 &\le O\left(\frac{1}{i^{\varkappa + 1}}\right) 
	 + C m \sqrt{ i} \left( i - m \sqrt{ i}  \right) ^{- p - 1} + C 2^p (2 i - 1)^{ -p - 1}\\
	 &= O\left(i^{- p - \frac{1}{2}}\right) 
.\end{align*}

\end{proof}

\begin{lemma}
	\[
		\operatorname{Var} \left[ \alpha(i, \tau_i^{\mathcal{B}} - i) \right] \le    O(i^{- 2 p - 1})
	.\] 
	\label{lemma:variance-diagonal}
\end{lemma}
\begin{proof}
	Since we have controlled expectation, it remains to control $\mathbb{E} [\alpha^2]$. Follow the same procedure, we obtain
\begin{align*}
	 \mathbb{E}\left[ \alpha\left(i, \tau_i^{\mathcal{B}} - i\right)^2  | G\right]
	 &\le \sup _{j:|i - j| < m \sqrt{i} } \alpha(i,j)^2 \\
	 &\le C \sup \left| 2^p B \left( \frac{1}{(2^i - 1)^p}  - \frac{1}{(2j)^p} + O(\frac{1}{i^{\varkappa
	 + 1}}) \right) \right|^2  \\
	 &= O(i^{-2p-1}) 
.\end{align*}
\end{proof}
